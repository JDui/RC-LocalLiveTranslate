cmake_minimum_required(VERSION 3.22)
project(rc_hymt LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ALL_VARIANTS OFF CACHE BOOL "" FORCE)
set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
add_subdirectory("${HY_SOURCE}" llama)
include(hy-stq-reader.cmake)
add_library(rc_hymt SHARED hy_jni.cpp)
target_link_libraries(rc_hymt PRIVATE llama)
target_compile_options(rc_hymt PRIVATE -fvisibility=hidden -fvisibility-inlines-hidden)
target_link_options(rc_hymt PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,--exclude-libs,ALL")
